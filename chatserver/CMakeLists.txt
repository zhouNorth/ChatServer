cmake_minimum_required(VERSION 3.16)
project(chatserver LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build in Linux/WSL using VS Code Remote - WSL.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/bin")
find_package(Threads REQUIRED)
set(MUDUO_ROOT "${PROJECT_SOURCE_DIR}/../.deps/muduo-install" CACHE PATH "muduo installation prefix")
find_path(MUDUO_INCLUDE_DIR muduo/net/TcpServer.h HINTS "${MUDUO_ROOT}/include")
find_library(MUDUO_NET_LIBRARY muduo_net HINTS "${MUDUO_ROOT}/lib" "${MUDUO_ROOT}/lib64")
find_library(MUDUO_BASE_LIBRARY muduo_base HINTS "${MUDUO_ROOT}/lib" "${MUDUO_ROOT}/lib64")
find_path(HIREDIS_INCLUDE_DIR hiredis/hiredis.h)
find_library(HIREDIS_LIBRARY hiredis)
find_path(MYSQL_INCLUDE_DIR mysql/mysql.h)
find_library(MYSQL_LIBRARY NAMES mysqlclient mariadb)
foreach(dependency HIREDIS_INCLUDE_DIR HIREDIS_LIBRARY MUDUO_INCLUDE_DIR MUDUO_NET_LIBRARY MUDUO_BASE_LIBRARY MYSQL_INCLUDE_DIR MYSQL_LIBRARY)
    if(NOT ${dependency})
        message(FATAL_ERROR "Missing ${dependency}. Run scripts/setup-linux.sh first.")
    endif()
endforeach()
add_subdirectory(src)
